\begin{goalbox}
\begin{itemize}
\item Cox 比例ハザードモデルを書き，回帰係数を対数ハザード比として解釈し，$S(t\mid x)=S_0(t)^{\exp(\beta x)}$ を導ける．

\end{itemize}
\end{goalbox}

\section{Cox 比例ハザードモデル}\label{sec:10-cox}
\Hissu\Hinshutsu
\subsection{モデル}
\begin{teigi}[Cox 比例ハザードモデル]\label{def:10-cox}
共変量 $\bm x$ をもつ個体のハザードを
\begin{equation}
h(t\mid\bm x)=h_0(t)\exp(\bm\beta\transp\bm x)\label{eq:10-cox}
\end{equation}
とする．$h_0(t)$ はベースラインハザード（$\bm x=\bm0$ のハザード）で，形を特定しない（セミパラメトリックモデル）．
\end{teigi}
2つの個体のハザード比は $h(t\mid\bm x_1)/h(t\mid\bm x_2)=\exp\{\bm\beta\transp(\bm x_1-\bm x_2)\}$ で $t$ によらない．
これが\emphx{比例ハザード性}である（\kakomon{2022}{1}〔1〕）．
2 値の共変量なら $e^{\beta}$ が HR，連続なら1単位増加あたりの HR である．

\begin{meidai}[生存関数の表現]\label[meidai]{prop:10-loglog}
$H_0(t)=\int_0^th_0$，$S_0(t)=e^{-H_0(t)}$ とすると
\begin{equation}
S(t\mid\bm x)=S_0(t)^{\exp(\bm\beta\transp\bm x)},\qquad
\log\{-\log S(t\mid\bm x)\}=\log H_0(t)+\bm\beta\transp\bm x.\label{eq:10-loglog}
\end{equation}
\end{meidai}
\begin{proof}
$H(t\mid\bm x)=\int_0^th_0(u)e^{\bm\beta\transp\bm x}\dd u=e^{\bm\beta\transp\bm x}H_0(t)$．
$S=e^{-H}$（\cref{thm:09-relations}）より $S(t\mid\bm x)=\{e^{-H_0(t)}\}^{\exp(\bm\beta\transp\bm x)}$．両辺の $-\log$ をとってから $\log$ をとれば第2式．
\end{proof}
第2式から，比例ハザードの下では群ごとの $\log\{-\log\hat S(t)\}$ の差が $\beta$ で一定になる（2重対数プロットが平行になる；\kakomon{2022}{1}〔3〕で与えられる）．

\paragraph{線形変換モデル（\kakomon{2017}{1}〔5〕）}
\cref{eq:10-loglog}を $t=T$ で評価すると $\log H_0(T)=-\bm\beta\transp\bm x+\epsilon$，$\epsilon=\log H(T\mid\bm x)$．
$H(T\mid\bm x)\sim\Ex(1)$（\cref{prop:09-HT}）なので $\Prob(\epsilon\le u)=\Prob\bigl(H(T\mid\bm x)\le e^u\bigr)=1-\exp(-e^u)$（極値分布）．
$V\sim\Unif(0,1)$ なら $\epsilon=\log(-\log V)$ として $T=H_0^{-1}\bigl(\exp\{-\bm\beta\transp\bm x+\epsilon\}\bigr)$ で乱数を作れる．

\subsection{部分尤度}\label{subsec:10-partial}
\Hissu 同順位のないイベント時点 $t_{(1)}<\dots<t_{(r)}$，時点 $t_{(j)}$ でイベントを起こした個体を $(j)$，リスク集合を $R_j$ とする．
\begin{teiri}[Cox の部分尤度]\label[teiri]{thm:10-partial}
\begin{equation}
L(\bm\beta)=\prod_{j=1}^r\frac{\exp(\bm\beta\transp\bm x_{(j)})}{\sum_{l\in R_j}\exp(\bm\beta\transp\bm x_l)}.\label{eq:10-partial}
\end{equation}
\end{teiri}
\begin{proof}[導出]
時点 $t_{(j)}$ で $R_j$ の中からちょうど1個体がイベントを起こしたという条件の下で，それが個体 $(j)$ である条件付き確率は
\[
\frac{h(t_{(j)}\mid\bm x_{(j)})}{\sum_{l\in R_j}h(t_{(j)}\mid\bm x_l)}=\frac{h_0(t_{(j)})e^{\bm\beta\transp\bm x_{(j)}}}{\sum_{l\in R_j}h_0(t_{(j)})e^{\bm\beta\transp\bm x_l}}
=\frac{e^{\bm\beta\transp\bm x_{(j)}}}{\sum_{l\in R_j}e^{\bm\beta\transp\bm x_l}}.
\]
$h_0$ が約分されて消える．これをすべてのイベント時点について掛け合わせたものが部分尤度である．
打ち切り例は分子には現れず，打ち切り時点まで分母のリスク集合に含まれる．
\end{proof}
\kakomon{2016}{3} では群2（$x=1$）の4つのイベント時点で分子 $e^\beta$，分母は（群1の人数）$+$（群2の人数）$e^\beta$ で，
\[
L(\beta)=\frac{e^\beta}{5+5e^\beta}\cdot\frac{e^\beta}{5+4e^\beta}\cdot\frac{e^\beta}{3+2e^\beta}\cdot\frac{e^\beta}{2+e^\beta}\cdot\frac12\cdot\frac11.
\]

\begin{meidai}[スコアと情報量]\label[meidai]{prop:10-score}
1次元の $\beta$ について，$\bar x_j(\beta)=\sum_{l\in R_j}x_le^{\beta x_l}/\sum_{l\in R_j}e^{\beta x_l}$（リスク集合での重み付き平均）とすると
\[
U(\beta)=\frac{\partial\log L}{\partial\beta}=\sum_j\{x_{(j)}-\bar x_j(\beta)\},\qquad
I(\beta)=-\frac{\partial^2\log L}{\partial\beta^2}=\sum_j\Bigl\{\frac{\sum_{R_j}x_l^2e^{\beta x_l}}{\sum_{R_j}e^{\beta x_l}}-\bar x_j(\beta)^2\Bigr\}.
\]
2 値の $x$ で $\beta=0$ とすると $U(0)=\sum_j(d_{1j}-n_{1j}/n_j)$，$I(0)=\sum_jn_{1j}n_{0j}/n_j^2$ で，
スコア検定 $U(0)^2/I(0)$ はログランク検定（\cref{thm:09-logrank}，同順位なし）に一致する．
\end{meidai}
部分尤度は通常の尤度と同様に扱える：$\hat\beta$ は漸近正規で $\SE(\hat\beta)=I(\hat\beta)^{-1/2}$，
Wald 検定 $\hat\beta/\SE$，部分尤度比検定 $2\{\log L(\hat\beta)-\log L(0)\}$ はいずれも漸近的に $\chi^2_1$（Wald は $\Normal(0,1)$）で近似できる．
\begin{supbox}[補足：同時点のイベント（Breslow 近似）]
同時点に複数のイベントがあるときは，Breslow 近似 $\prod_j\exp(\bm\beta\transp\bm s_j)/\{\sum_{R_j}e^{\bm\beta\transp\bm x_l}\}^{d_j}$（$\bm s_j$ はその時点でイベントを起こした個体の共変量の和）がよく用いられる．
2 値の $x$ では $U(0)=\sum_j(d_{1j}-d_jn_{1j}/n_j)$ はログランクの $O-E$ に一致し，$I(0)=\sum_jd_jn_{1j}n_{0j}/n_j^2$ は超幾何分散 $V_j$ より因子 $(n_j-1)/(n_j-d_j)$ だけ大きい（$d_j=1$ なら一致）．
\end{supbox}

\subsection{推定結果の解釈}\label{subsec:10-interp}
\kakomon{2022}{1} の数値では，$\hat\beta=0.06$，$\SE=0.18$，$z_{0.025}=1.96$ から $1.96\times0.18=0.353$ で
\[
\widehat{\HR}=e^{0.06}=1.06,\qquad 95\%\text{CI}=(e^{0.06-0.353},\,e^{0.06+0.353})=(0.75,\,1.51).
\]
HR $<1$ なら試験群のハザードが低い（有効），$>1$ なら高い．点推定値はプラセボ群がわずかに良い方向だが，
区間が1を含むので，死亡までの時間に差があるとは言えない．

\section{パラメトリックモデル：指数分布}\label{sec:10-exp}
\Hissu\Hinshutsu
\subsection{打ち切りのある尤度}
\begin{teiri}[打ち切りのある尤度（\kakomon{2018}{1}〔2〕）]\label[teiri]{thm:10-censlik}
独立な右側打ち切りの下で，観測 $(t_i,\delta_i)$ の尤度は
\begin{equation}
L=\prod_{i=1}^nf(t_i)^{\delta_i}S(t_i)^{1-\delta_i}=\prod_{i=1}^nh(t_i)^{\delta_i}S(t_i).\label{eq:10-censlik}
\end{equation}
\end{teiri}
イベント例は「その時点で起きた」（密度），打ち切り例は「その時点より後に起きる」（生存関数）を寄与する．
打ち切りの分布はパラメータを含まないとして尤度から除いている（独立で無情報な打ち切り）．

\begin{meidai}[指数モデルの MLE と情報量]\label[meidai]{prop:10-expmle}
$h(t)=\lambda$ のとき，$d=\sum\delta_i$（イベント数），$W=\sum t_i$（総観察時間，人時間）として
\begin{equation}
\log L=d\log\lambda-\lambda W,\qquad \hat\lambda=\frac dW,\qquad I(\lambda)=-\frac{\partial^2\log L}{\partial\lambda^2}=\frac{d}{\lambda^2}.\label{eq:10-expmle}
\end{equation}
\end{meidai}
\begin{proof}
$f=\lambda e^{-\lambda t}$，$S=e^{-\lambda t}$ を\cref{eq:10-censlik}に代入すると $L=\prod\lambda^{\delta_i}e^{-\lambda t_i}=\lambda^de^{-\lambda W}$．
$\partial\log L/\partial\lambda=d/\lambda-W=0$ から $\hat\lambda$．2階微分は $-d/\lambda^2$．
\end{proof}
$\hat\lambda$ は「イベント数/人時間」の発生率そのものである．打ち切りがなければ $d=n$，$\hat\lambda=1/\bar T$，$I_n(\lambda)=n/\lambda^2$（\kakomon{2023}{2}）．
\Youchui 打ち切りがあるとき，期待情報量は $\E[d]/\lambda^2$ で，$\E[d]$ は打ち切り分布に依存する．実用上は観測情報 $d/\hat\lambda^2$ を用いる．
\kakomon{2019}{1} のデータ（イベント3，総観察時間83週）では $\hat\lambda=3/83=0.036$/週．

\subsection{デルタ法とハザード比}\label{subsec:10-expdelta}
$\hat\lambda$ は漸近的に $\V[\hat\lambda]\approx\lambda^2/d$ なので，デルタ法で $\V[\log\hat\lambda]\approx\lambda^{-2}\cdot\lambda^2/d=1/d$．
独立な2群では
\begin{equation}
\V[\log\hat\lambda_1-\log\hat\lambda_2]\approx\frac{1}{d_1}+\frac{1}{d_2},\qquad
Z=\frac{\log\hat\lambda_1-\log\hat\lambda_2}{\sqrt{1/d_1+1/d_2}}.\label{eq:10-expz}
\end{equation}
$\log$ 変換後の分散が\textbf{イベント数だけ}で決まることが重要である．生存時間試験の精度は症例数ではなくイベント数で決まる．
指数モデルの HR $=\lambda_1/\lambda_2$ の95\%CI は $\exp\{\log(\hat\lambda_1/\hat\lambda_2)\pm1.96\sqrt{1/d_1+1/d_2}\}$．

\paragraph{必要イベント数（\kakomon{2018}{1}〔5〕）}
各群のイベント数を $r$ とすると，対立仮説の下で $Z$ はおよそ $\Normal\bigl(\frac{\log\lambda_1-\log\lambda_2}{\sqrt{2/r}},1\bigr)$ で，
両側 $\alpha$・検出力 $1-\beta$ の条件 $\frac{|\log\lambda_1-\log\lambda_2|}{\sqrt{2/r}}=z_{\alpha/2}+z_\beta$ から
\begin{equation}
r=\frac{2(z_{\alpha/2}+z_\beta)^2}{(\log\lambda_1-\log\lambda_2)^2}\quad\text{（各群）}.\label{eq:10-events}
\end{equation}
導出の一般論は\cref{ch:12}．

\section{競合リスクの解釈}\label{sec:10-compete}
\Hissu
がん死を評価項目とする研究で他因死が起こると，その人のがん死は以後決して観察されない．
このように，あるイベントが起こると別のイベントが起こりえなくなるとき，それらを\emphx{競合リスク}という．
\Youchui 打ち切りは「イベントはいずれ起こるが観察できなかった」，競合イベントは「そのイベントは二度と起こらない」であり，両者は別物である．
本節では解釈に必要な最小限の定義を置き，2つの論点（$1-\mathrm{KM}$ の過大推定，原因別 Cox の係数の意味）を説明する．

\subsection{定義}\label{subsec:10-compete-def}
最初のイベントまでの時間を $T$，その種類（原因）を $J\in\{1,\dots,m\}$ とする（同時には起こらないとする）．$S(t)=\Prob(T>t)$ は「どのイベントもまだ起きていない」確率である．
\begin{teigi}[原因別ハザード・累積発生関数・部分分布ハザード]\label{def:10-compete}
\begin{align*}
&\text{原因別ハザード} & \lambda_j(t)&=\lim_{\Delta\to0}\frac{\Prob(t\le T<t+\Delta,\ J=j\mid T\ge t)}{\Delta},\qquad \Lambda_j(t)=\int_0^t\lambda_j(u)\,\dd u,\\
&\text{累積発生関数（CIF）} & F_j(t)&=\Prob(T\le t,\ J=j),\\
&\text{部分分布ハザード} & \tilde\lambda_j(t)&=-\frac{\dd}{\dd t}\log\{1-F_j(t)\}.
\end{align*}
\end{teigi}
事象 $\{t\le T<t+\Delta\}$ は排反な $\{t\le T<t+\Delta,J=j\}$ $(j=1,\dots,m)$ の和なので，全ハザードは $\lambda(t)=\sum_j\lambda_j(t)$，
したがって\cref{thm:09-relations}より
\begin{equation}
S(t)=\exp\Bigl\{-\sum_{j=1}^m\Lambda_j(t)\Bigr\},\qquad
F_j(t)=\int_0^t\lambda_j(u)S(u)\,\dd u .\label{eq:10-cif}
\end{equation}
第2式は「$u$ までどのイベントも起きず（確率 $S(u)$），そこで原因 $j$ が起こる（率 $\lambda_j(u)$）」を足し上げたもので，$\Prob(u\le T<u+\dd u,J=j)=\lambda_j(u)S(u)\,\dd u$ による．
$F_j$ は $S$ を通じて\textbf{他の原因のハザードにも依存する}．また $\sum_jF_j(t)=1-S(t)$ で，$F_j(\infty)=\Prob(J=j)<1$（$m\ge2$ で他の原因の確率が正のとき）である．

部分分布ハザードについては，$1-F_j(t)=\Prob(T>t)+\Prob(T\le t,\,J\ne j)$ なので，
$\tilde\lambda_j$ は\textbf{他の原因で既にイベントを起こした人もリスク集合に残したときの原因 $j$ の瞬間発生率}に相当する．
定義から $1-F_j(t)=\exp\{-\int_0^t\tilde\lambda_j(u)\,\dd u\}$ であり，$\tilde\lambda_j$ と $F_j$ は1対1に対応する．

\subsection{\texorpdfstring{$1-\mathrm{KM}$}{1−KM} による過大推定}\label{subsec:10-compete-km}
\Hinshutsu 他の原因のイベントを打ち切りとみなし，原因 $j$ だけをイベントとして KM $\hat S_j^{\mathrm{KM}}$ を計算するとする．
各時点の因子 $1-d_{ji}/n_i$ の $n_i$ は「どのイベントも起こさず観察中の人数」なので，$d_{ji}/n_i$ は原因別ハザード $\lambda_j$ の増分を推定しており，
$\hat S_j^{\mathrm{KM}}(t)$ は（独立な打ち切りの下で）$\exp\{-\Lambda_j(t)\}$ の推定量になる．したがって $1-\hat S_j^{\mathrm{KM}}(t)$ が推定するのは $1-e^{-\Lambda_j(t)}$ であって $F_j(t)$ ではない．
\begin{meidai}[$1-\mathrm{KM}$ の推定対象は CIF 以上]\label[meidai]{prop:10-overestimate}
$F_j(t)\le1-e^{-\Lambda_j(t)}$．$\lambda_j(u)>0$ かつ $\sum_{k\ne j}\Lambda_k(u)>0$ となる $u$ が $[0,t]$ 内の正の長さの区間にわたって存在すれば，不等号は厳密である．
\end{meidai}
\begin{proof}
$S(u)=e^{-\Lambda_j(u)}\exp\{-\sum_{k\ne j}\Lambda_k(u)\}\le e^{-\Lambda_j(u)}$ を\cref{eq:10-cif}に用いて
$F_j(t)=\int_0^t\lambda_j(u)S(u)\,\dd u\le\int_0^t\lambda_j(u)e^{-\Lambda_j(u)}\,\dd u=\bigl[-e^{-\Lambda_j(u)}\bigr]_0^t=1-e^{-\Lambda_j(t)}$．
\end{proof}
\paragraph{解釈（\kakomon{2021}{1}〔5〕）}
打ち切り扱いは「他因死した人も，リスク集合に残った人と同じ率で将来原因 $j$ のイベントを起こす」とみなすことにあたる．
実際には他因死した人は原因 $j$ のイベントを二度と起こさないので，原因 $j$ のイベントを起こさない確率を過小に，累積発生を過大に推定する．
$1-e^{-\Lambda_j(t)}$ は「他の原因が存在しない仮想的な世界での原因 $j$ の累積リスク」と解釈されることがあるが，それには原因ごとの潜在的なイベント時間が独立という，データからは検証できない仮定が要る．
現実の世界で「$t$ までに原因 $j$ のイベントを経験する確率」は $F_j(t)$ である．

\subsection{原因別 Cox と Fine--Gray モデル}\label{subsec:10-compete-reg}
\Hinshutsu
\paragraph{原因別ハザードの Cox モデル}
$\lambda_1(t\mid x)=\lambda_{10}(t)\exp(\beta x)$ とする．原因別ハザードのリスク集合は「どのイベントも起きていない人」なので，
他の原因のイベントを打ち切りとみなした通常の Cox 回帰（\cref{sec:10-cox}）で $\beta$ を推定できる．
しかし $e^\beta$ は\textbf{原因別ハザード $\lambda_1$ の比}（まだどのイベントも起きていない人での原因1の発生率の比）であり，$F_1$ への効果ではない（\kakomon{2025}{3}〔5〕）．
実際，\cref{eq:10-cif}より
\[
F_1(t\mid x)=\int_0^t\lambda_1(u\mid x)\exp\{-\Lambda_1(u\mid x)-\Lambda_2(u\mid x)\}\,\dd u
\]
は $\lambda_2(\cdot\mid x)$ にも依存する．$x$ が $\lambda_2$ を大きくすれば，$\beta$ の値にかかわらず $F_1$ は小さくなりうる．
$x$ が $\lambda_2$ に影響しない場合でも，$F_1$ は $\beta$ の非線形な関数で，$e^\beta$ は $F_1$ の比などにはならない．

\paragraph{Fine--Gray モデル（部分分布ハザードの比例ハザードモデル）}
$F_1$ への効果を直接表すには，部分分布ハザードに比例ハザードを仮定する \cite{finegray1999}：
\[
\tilde\lambda_1(t\mid x)=\tilde\lambda_{10}(t)\exp(\gamma x)
\quad\Longrightarrow\quad
1-F_1(t\mid x)=\exp\Bigl\{-e^{\gamma x}\int_0^t\tilde\lambda_{10}\Bigr\}=\{1-F_{10}(t)\}^{\exp(\gamma x)}.
\]
ここで $F_{10}$ は $x=0$ の CIF である．$\gamma>0$ なら $0<F_{10}(t)<1$ となるすべての $t$ で $F_1(t\mid x=1)>F_1(t\mid x=0)$ となり，係数を「累積発生を増やす（$\gamma<0$ なら減らす）」と直接解釈できる．
ただし $e^\gamma$ も $F_1$ の比そのものではなく，部分分布ハザードの比である．
\begin{pointbox}[本書での扱い]
Fine--Gray モデルは\textbf{係数の解釈（原因別 Cox との違い）だけを必修}とし，推定法は扱わない．
病因の研究（まだイベントのない人で原因 $j$ の発生率が変わるか）には原因別ハザード，
予後の予測（$t$ までに原因 $j$ のイベントを経験する確率が変わるか）には CIF と Fine--Gray の係数が対応する．
\end{pointbox}

\section{仮定}\label{sec:10-assump}
\begin{itemize}
\item Cox：比例ハザード（HR が時間によらず一定），共変量の効果が $\exp(\bm\beta\transp\bm x)$ の形（対数線形），独立な打ち切り．
\item 指数モデル：ハザード一定，独立で無情報な打ち切り．
\item 競合リスク：独立な打ち切りは追跡不能・試験終了など真の打ち切りに対する仮定であり，\textbf{競合イベントは打ち切りではない}．
      原因別ハザードと CIF は観測データから推定できるが，「競合イベントがない世界での分布」は検証できない仮定なしには推定できない．
\end{itemize}

\section{結果の解釈}\label{sec:10-interp}
\begin{itemize}
\item 「HR $=0.70$（95\%CI 0.55--0.89）」：追跡期間中どの時点でも，生存している人の死亡の瞬間的な発生率が試験群で30\%低い．
      「死亡割合が30\%低い」ではない（\cref{subsec:02-relation}）．
\item 共変量で調整した HR は「他の共変量が同じ人どうしの比較」．Cox の HR も OR と同様に非可縮であり，RCT でも調整で値が変わりうる．
\item 治療が原因1の原因別ハザードを下げても，生存が延びた結果として原因2のイベントが増えることがある．競合リスクがあるときは両方の原因について示す．
\end{itemize}

\section{手法選択}\label{sec:10-choice}
\begin{itemize}
\item 2群比較の検定 → ログランク（\cref{sec:09-logrank}）．効果の大きさ・共変量調整 → Cox 回帰．
\item ハザード一定がもっともらしく小標本・外挿が必要 → 指数モデル．
\item 比例ハザードが疑わしい → HR 1つで要約せず，群ごとの KM 曲線や特定時点の $\hat S(t)$ を示す．
\item 競合リスクがある → 原因別ハザードへの効果は原因別 Cox（競合を打ち切り扱い），累積発生への効果の解釈は Fine--Gray モデルの係数．
      累積発生を $1-\mathrm{KM}$（競合を打ち切り扱い）で報告しない．
\end{itemize}

\section{よくある誤り}\label{sec:10-err}
\begin{warnbox}
\begin{itemize}
\item 部分尤度の分母を，その時点のリスク集合（両群を含め，直前までイベントも打ち切りも起きていない全員）ではなく同じ群の人だけとする．打ち切り例を分母から早く外しすぎる．
\item 打ち切り例の尤度の寄与を $f(t_i)$ とする（正しくは $S(t_i)$）．
\item 指数モデルの分散 $\V[\log\hat\lambda]$ を $1/n$ とする（正しくは $1/d$，イベント数）．
\item 競合イベントを打ち切りとして $1-\mathrm{KM}$ で累積発生を報告する（過大推定）．
\item 原因別ハザード比を「累積発生の比」と解釈する．$F_j(t)\to1$ と考える（極限は $\Prob(J=j)$）．
\end{itemize}
\end{warnbox}

\section{数理統計との接続}\label{sec:10-link}
\begin{linkbox}
\begin{itemize}
\item 指数分布の MLE・Fisher情報（打ち切りなし）は『現代数理統計学の基礎』第6・7章演習（指数分布の片側検定と検出力，2群の指数分布の比較）と同じ計算で，
      打ち切りがあると $n$ が $d$ に，$\sum T_i$ が人時間 $W$ に置き換わる．
      \kakomonSM{2016}{2} は指数分布の MLE・不変性・$\sum X_i\sim\Ga(n,\lambda)$（形状・率）・生存関数の UMVU を問うた．
\item 指数モデルの尤度 $\lambda^de^{-\lambda W}$ は，平均 $\lambda W$ のポアソン分布で $d$ を観測した尤度と $\lambda$ について同じ形である（\cref{ch:17}）．
\item デルタ法（同書 定理5.20）で $\V[\log\hat\lambda]=1/d$．$\log$ 変換は分散を $\lambda$ に依存しなくする分散安定化変換でもある．
\item 部分尤度は条件付き確率の積であり，局外母数 $h_0$ を消去する点で層別 $2\times2$ 表の条件付け（\cref{sec:04-mh}）と同じ発想である．
\item 独立な $\Ex(\lambda)$ と $\Ex(\mu)$ の最小値 $Z$ と，どちらが最小かの指示変数 $W$ の同時分布（同書第4章演習）は，一定の原因別ハザードをもつ競合リスクそのものである：
      $Z\sim\Ex(\lambda+\mu)$，$\Prob(W=1)=\lambda/(\lambda+\mu)$，$Z\indep W$．
\end{itemize}
\end{linkbox}

\section{例題}\label{sec:10-ex}
\begin{reidai}[部分尤度とスコア検定]\label{reidai:10-1}
5例のデータ（時間，イベント有無，群 $x$）：$(2,\text{イベント},1)$，$(3,\text{イベント},0)$，$(4,\text{打ち切り},1)$，$(5,\text{イベント},0)$，$(6,\text{イベント},1)$．
\begin{enumerate}[label=(\arabic*)]
\item Cox モデル $h(t\mid x)=h_0(t)e^{\beta x}$ の部分尤度 $L(\beta)$ を書け．
\item $\beta=0$ でのスコア $U(0)$ と情報量 $I(0)$ を求め，スコア検定統計量を計算せよ．
\item (2)がログランク検定統計量に一致することを，時点ごとの $O-E$ と分散から確かめよ．
\end{enumerate}
\end{reidai}

\begin{reidai}[指数モデルによる2群比較]\label{reidai:10-2}
群Aは総観察時間240人月でイベント12件，群Bは200人月で20件であった．指数分布を仮定する．
\begin{enumerate}[label=(\arabic*)]
\item 各群のハザードの MLE と中央生存時間を求めよ（$\log2=0.6931$）．
\item ハザード比 $\lambda_A/\lambda_B$ の推定値と，\cref{eq:10-expz}の $Z$ 検定（両側5\%）を行え．
\item HR の95\%信頼区間を求めよ（$e^{-1.4088}=0.244$，$e^{0.0225}=1.023$）．
\end{enumerate}
\end{reidai}

\begin{reidai}[比例ハザードの下での生存関数]\label{reidai:10-4}
対照群の生存関数が $S_0(t)=\exp\{-(t/10)^{1.5}\}$ で，試験群は対照群に対するハザード比 $0.6$ の比例ハザードモデルに従うとする．
\begin{enumerate}[label=(\arabic*)]
\item 試験群の生存関数を書け．
\item $t=10$ での両群の生存確率を求めよ（$e^{-1}=0.368$，$e^{-0.6}=0.549$）．
\item $t=10$ までのイベント割合の比 $\{1-S_1(10)\}/\{1-S_0(10)\}$ を求め，HR $=0.6$ と比べよ．
\end{enumerate}
\end{reidai}

\begin{reidai}[競合リスク：一定の原因別ハザード]\label{reidai:10-5}
原因1・原因2の原因別ハザードが（年あたり）それぞれ $\lambda_1=0.02$，$\lambda_2=0.03$ で一定とする．
$e^{-0.5}=0.6065$，$e^{-0.2}=0.8187$，$e^{-1}=0.3679$ を用いよ．
\begin{enumerate}[label=(\arabic*)]
\item 10年間いずれのイベントも起こさない確率 $S(10)$ を求めよ．
\item $F_1(10)$ と $F_1(\infty)$ を求めよ．
\item 原因2を打ち切りとみなした $1-\mathrm{KM}$ が推定する量 $1-e^{-\Lambda_1(10)}$ を求め，$F_1(10)$ と比べよ．
\item ある共変量 $x=1$ の集団では $\lambda_1=0.02$ のまま $\lambda_2=0.08$ であった．原因1の原因別ハザード比は1である．この集団の $F_1(10)$ を求め，原因別ハザード比で $F_1$ への効果を評価することの問題点を述べよ．
\end{enumerate}
\end{reidai}

\section{解答}\label{sec:10-sol}
\begin{kaitou}{reidai:10-1}
(1) リスク集合：$t=2$：全5例（$x=1,0,1,0,1$），$t=3$：4例（$0,1,0,1$），$t=5$：2例（$0,1$），$t=6$：1例（$1$）．
\[
L(\beta)=\frac{e^\beta}{3e^\beta+2}\cdot\frac{1}{2e^\beta+2}\cdot\frac{1}{e^\beta+1}\cdot\frac{e^\beta}{e^\beta}
=\frac{e^\beta}{(3e^\beta+2)(2e^\beta+2)(e^\beta+1)}.
\]
(2) $\beta=0$ で $\bar x_j=3/5,\ 2/4,\ 1/2,\ 1$．$U(0)=(1-0.6)+(0-0.5)+(0-0.5)+(1-1)=-0.6$．
$I(0)=\sum\bar x_j(1-\bar x_j)=0.24+0.25+0.25+0=0.74$．スコア統計量 $0.36/0.74=0.49$（$<3.841=\chi^2_1(0.05)$）．\\
(3) 群1（$x=1$）の観測イベント2，期待 $0.6+0.5+0.5+1=2.6$，$O-E=-0.6$．各時点 $d_j=1$ なので分散 $\sum n_{1j}n_{0j}/n_j^2=0.74$．一致する．
\end{kaitou}

\begin{kaitou}{reidai:10-2}
(1) $\hat\lambda_A=12/240=0.05$/月，$\hat\lambda_B=20/200=0.10$/月．中央値 $\log2/\hat\lambda$：A 13.9か月，B 6.9か月．\\
(2) $\widehat{\HR}=0.5$，$\log0.5=-0.693$，$\SE=\sqrt{1/12+1/20}=\sqrt{0.1333}=0.365$，$Z=-1.90$，$|Z|<z_{0.025}=1.96$ で有意でない．\\
(3) $-0.693\pm1.96\times0.365=(-1.409,\,0.023)$ → $(0.24,\,1.02)$．区間が1をわずかに含む．
\end{kaitou}

\begin{kaitou}{reidai:10-4}
(1) \cref{prop:10-loglog}より $S_1(t)=S_0(t)^{0.6}=\exp\{-0.6(t/10)^{1.5}\}$．\\
(2) $S_0(10)=e^{-1}=0.368$，$S_1(10)=e^{-0.6}=0.549$．\\
(3) $(1-0.549)/(1-0.368)=0.451/0.632=0.71$．HR $=0.6$ とは異なる．HR は瞬間的な発生率の比であり，一定期間のイベント割合の比（リスク比）ではない．
\end{kaitou}

\begin{kaitou}{reidai:10-5}
(1) 全ハザード $\lambda=0.05$ なので $S(10)=e^{-0.5}=0.607$．\\
(2) $F_1(t)=\int_0^t0.02\,e^{-0.05u}\dd u=\frac{0.02}{0.05}(1-e^{-0.05t})$ より $F_1(10)=0.4\times0.393=0.157$，$F_1(\infty)=0.4$．\\
(3) $\Lambda_1(10)=0.2$ で $1-e^{-0.2}=0.181>0.157$．$1-\mathrm{KM}$ の推定対象は $F_1(10)$ を約15\%上回る（\cref{prop:10-overestimate}）．\\
(4) 全ハザード $0.10$ で $F_1(10)=\frac{0.02}{0.10}(1-e^{-1})=0.2\times0.632=0.126<0.157$．
原因1の原因別ハザード比は1（$\beta=0$）なのに，原因2のハザードが大きいため原因1のイベントを起こす前に原因2で脱落する人が増え，$F_1$ は小さくなる．
$F_1$ は $\lambda_2$ にも依存するので，原因別ハザード比だけでは $F_1$ への効果を評価できない．
\end{kaitou}

\begin{summarybox}
\begin{itemize}
\item Cox：$h(t\mid\bm x)=h_0(t)e^{\bm\beta\transp\bm x}$，HR $=e^\beta$ は時間によらず一定．$S=S_0^{\exp(\bm\beta\transp\bm x)}$．
\item 部分尤度 $\prod_j e^{\bm\beta\transp\bm x_{(j)}}/\sum_{R_j}e^{\bm\beta\transp\bm x_l}$．$\beta=0$ のスコア検定＝ログランク．
\item 打ち切り尤度 $\prod f^{\delta}S^{1-\delta}$．指数：$\hat\lambda=d/W$，$I=d/\lambda^2$，$\V[\log\hat\lambda]\approx1/d$．
\item 2群：$\V[\log\hat\lambda_1-\log\hat\lambda_2]\approx1/d_1+1/d_2$，必要イベント数 $r=2(z_{\alpha/2}+z_\beta)^2/(\log\HR)^2$（各群）．
\item 競合リスク：$\lambda=\sum\lambda_j$，$S=\exp(-\sum\Lambda_j)$，$F_j(t)=\int_0^t\lambda_jS$，$\tilde\lambda_j=-\frac{\dd}{\dd t}\log(1-F_j)$．
\item 競合を打ち切り扱いした $1-\mathrm{KM}$ は $1-e^{-\Lambda_j}\ge F_j$ を推定し過大．
\item 原因別 Cox の $\beta$ は原因別ハザードへの効果で CIF への効果ではない．CIF への効果は Fine--Gray（部分分布ハザード）の係数で解釈する（推定法は扱わない）．
\end{itemize}
\end{summarybox}
