\chapter{検出力・症例数・イベント数設計}\label{ch:12}

\begin{goalbox}
\begin{itemize}
\item 第1種の過誤・第2種の過誤・検出力・効果量の関係を説明できる．
\item 「帰無仮説で $\Normal(0,1)$，対立仮説で $\Normal(\mu,1)$」の枠組みから $\mu=z_\alpha+z_\beta$ を導出し，正規近似できる検定の症例数の式を作れる．
\item 平均の差・比率の差の症例数を公式から計算できる．
\item 生存時間試験では必要な\textbf{イベント数}が精度を決めることを説明し，指数モデルとログランク検定（Schoenfeld の式）からイベント数を導き，症例数に換算できる．
\item 分散を帰無仮説の下で評価するか対立仮説の下で評価するかで結果が変わることを理解する．
\end{itemize}
\end{goalbox}

\begin{itemize}

\section{問題設定と基本概念}\label{sec:12-basic}
\begin{center}\small
\begin{tabular}{@{}lcc@{}}
\toprule
 & $H_0$ が真 & $H_1$ が真\\
\midrule
$H_0$ を棄却 & 第1種の過誤（確率 $\alpha$） & 正しい（検出力 $1-\beta$）\\
$H_0$ を採択 & 正しい（$1-\alpha$） & 第2種の過誤（確率 $\beta$）\\
\bottomrule
\end{tabular}
\end{center}
症例数設計では，(i) 有意水準 $\alpha$（片側・両側），(ii) 検出したい効果量 $\Delta$（臨床的に意味のある最小の差，または期待される差），
(iii) 分散などの局外母数の見積もり，(iv) 目標検出力 $1-\beta$（通常0.8または0.9）を定め，必要な $n$ を求める．
\Youchui 『現代数理統計学の基礎』では $\beta(\theta)$ を検出力関数そのものとして用いる．本書（と医療統計の慣習）では $\beta$ は第2種の過誤確率である．

\section{数理：\texorpdfstring{$\mu=z_\alpha+z_\beta$}{μ = zα + zβ} の導出}\label{sec:12-core}
\Hissu
\begin{teiri}[基本公式]\label[teiri]{thm:12-core}
検定統計量 $Z$ が $H_0$ の下で $\Normal(0,1)$，$H_1$ の下で $\Normal(\mu,1)$（$\mu>0$）に従い，$Z>z_\alpha$ で棄却するとき，
検出力は $\Phi(\mu-z_\alpha)$ であり，検出力 $1-\beta$ を達成する条件は
\begin{equation}
\mu=z_\alpha+z_\beta.\label{eq:12-core}
\end{equation}
\end{teiri}
\begin{proof}
$1-\beta=\Prob(Z>z_\alpha\mid H_1)=\Prob(Z-\mu>z_\alpha-\mu)=1-\Phi(z_\alpha-\mu)=\Phi(\mu-z_\alpha)$．
一方 $\Phi(z_\beta)=1-\beta$ なので，$\Phi$ の単調性から $\mu-z_\alpha=z_\beta$．
\end{proof}

両側検定（$|Z|>z_{\alpha/2}$）では，反対側の棄却確率（$\mu$ が大きければほぼ0）を無視して $\mu=z_{\alpha/2}+z_\beta$ とする（近似）．

\paragraph{症例数の一般公式}
推定量 $\hat\theta$ が近似的に $\hat\theta\sim\Normal(\theta,v/n)$（$v$ は1単位あたりの分散）で，$H_0:\theta=\theta_0$，対立仮説の値 $\theta_1$ に対して
$Z=(\hat\theta-\theta_0)/\sqrt{v/n}$ を用いるなら $\mu=(\theta_1-\theta_0)\sqrt{n/v}$ だから
\begin{equation}
n=\frac{(z_\alpha+z_\beta)^2\,v}{(\theta_1-\theta_0)^2}.\label{eq:12-general}
\end{equation}
分散が $H_0$ と $H_1$ で異なり（$v_0,v_1$），棄却限界を $H_0$ の分散で決めるなら
\begin{equation}
n=\frac{\bigl(z_\alpha\sqrt{v_0}+z_\beta\sqrt{v_1}\bigr)^2}{(\theta_1-\theta_0)^2}.\label{eq:12-general2}
\end{equation}
（$\Prob(\hat\theta>\theta_0+z_\alpha\sqrt{v_0/n}\mid\theta_1)=\Phi\bigl\{\bigl((\theta_1-\theta_0)\sqrt n-z_\alpha\sqrt{v_0}\bigr)/\sqrt{v_1}\bigr\}=1-\beta$ を $n$ について解く．）

\section{主な症例数の式}\label{sec:12-formulas}
\subsection{平均の差・比率の差}
準1級の範囲なので公式だけをまとめる．いずれも\cref{eq:12-general,eq:12-general2}に分散を代入したものである．
\begin{itemize}
\item 2群の平均の差（各群 $n$，共通 SD $\sigma$，差 $\Delta$）：$\V[\bar Y_1-\bar Y_0]=2\sigma^2/n$ より $v=2\sigma^2$ で
\begin{equation}
n=\frac{2\sigma^2(z_{\alpha/2}+z_\beta)^2}{\Delta^2}\quad\text{（各群）}.\label{eq:12-means}
\end{equation}
\item 対応のある比較（差の SD $\sigma_D$）：$n=\sigma_D^2(z_{\alpha/2}+z_\beta)^2/\Delta^2$（対の数）．
\item 2群の比率 $\pi_1,\pi_0$（各群 $n$，$\bar\pi=(\pi_1+\pi_0)/2$）：プール分散による検定（Pearson $\chi^2$ 検定と同等）では $v_0=2\bar\pi(1-\bar\pi)$，$v_1=\pi_1(1-\pi_1)+\pi_0(1-\pi_0)$ で
\begin{equation}
n=\frac{\bigl\{z_{\alpha/2}\sqrt{2\bar\pi(1-\bar\pi)}+z_\beta\sqrt{\pi_1(1-\pi_1)+\pi_0(1-\pi_0)}\bigr\}^2}{(\pi_1-\pi_0)^2}\quad\text{（各群）}.\label{eq:12-prop}
\end{equation}
簡便には $v_0$ も $v_1$ で置き換えて $n=(z_{\alpha/2}+z_\beta)^2\{\pi_1(1-\pi_1)+\pi_0(1-\pi_0)\}/(\pi_1-\pi_0)^2$．
\item 小標本で $t$ 検定を用いる場合は $z$ を $t$ に置き換えて反復計算する（数人増える程度）．
\end{itemize}

\subsection{生存時間：イベント数}\label{subsec:12-events}
\Hinshutsu
\paragraph{指数モデル（\kakomon{2018}{1}）}
群 $g$ の生存時間が率 $\lambda_g$ の指数分布に従い，イベント数が $d_g$ のとき，打ち切りを含む尤度から $\hat\lambda_g=d_g/\sum t_i$，Fisher 情報量 $I(\lambda_g)=d_g/\lambda_g^2$（\cref{ch:10}）．
デルタ法により $\V[\log\hat\lambda_g]\approx(1/\lambda_g)^2\cdot\lambda_g^2/d_g=1/d_g$．
2群は独立なので，各群 $r$ イベントなら $\V[\log\hat\lambda_1-\log\hat\lambda_0]\approx2/r$ であり，
$Z=(\log\hat\lambda_1-\log\hat\lambda_0)/\sqrt{2/r}$ は $H_1$ の下で近似的に $\Normal\bigl(\log\HR/\sqrt{2/r},1\bigr)$（$\HR=\lambda_1/\lambda_0$）．
両側検定で $|\log\HR|/\sqrt{2/r}=z_{\alpha/2}+z_\beta$（\cref{thm:12-core}）とおけば
\begin{equation}
r=\frac{2(z_{\alpha/2}+z_\beta)^2}{(\log\HR)^2}\quad\text{（各群）}.\label{eq:12-exp}
\end{equation}

\paragraph{ログランク検定（Schoenfeld の式）}
群1への割付割合を $p$，総イベント数を $D$ とし，同時イベントはない（各イベント時点のイベント数 $d_j=1$）とする．
ログランク統計量 $U=O_1-E_1$ とその分散 $V=\sum_jV_j$（\cref{thm:09-logrank}）は，$d_j=1$ のとき
\[
V_j=\frac{n_{1j}}{n_j}\Bigl(1-\frac{n_{1j}}{n_j}\Bigr)\qquad(j=1,\dots,D)
\]
となる．
\begin{meidai}[Schoenfeld の近似]\label[meidai]{prop:12-schoenfeld}
ハザード比が1に近く（$H_0$ の近傍），両群の打ち切りの状況が同じとき，近似的に
\[
V\approx Dp(1-p),\qquad \V[\log\widehat{\HR}]\approx\frac{1}{Dp(1-p)},\qquad
Z=\frac{U}{\sqrt V}\ \approx\ \Normal\Bigl(\log\HR\,\sqrt{Dp(1-p)},\ 1\Bigr).
\]
\end{meidai}
\begin{proof}[導出（近似）]
(i) $H_0$ の近傍では両群のハザードがほぼ等しく，打ち切りも両群で同様なので，リスク集合の構成比は割付比のまま保たれ $n_{1j}/n_j\approx p$ である．
よって $V=\sum_{j=1}^DV_j\approx\sum_{j=1}^Dp(1-p)=Dp(1-p)$．\\
(ii) ログランク検定は，Cox モデル $\lambda(t\mid x)=\lambda_0(t)e^{\beta x}$（群1で $x=1$）の $H_0:\beta=0$ に対するスコア検定であり（\cref{ch:10}），
$U$ は $\beta=0$ でのスコア $U(0)$，$V$ は情報量 $I(0)$ に一致する（同時イベントがない場合）．
スコアの1次のテイラー展開 $U(\beta)\approx U(0)-I(0)\beta$ に $\beta=\hat\beta$（$U(\hat\beta)=0$）を代入すると $\hat\beta\approx U/V$．
$\V[U]\approx V$ なので
\[
\V[\log\widehat{\HR}]=\V[\hat\beta]\approx\frac{V}{V^2}=\frac1V\approx\frac1{Dp(1-p)}.
\]
(iii) 真の値を $\beta=\theta=\log\HR$ とすると，同じ展開で $U(0)\approx U(\theta)+V\theta$，$\E[U(\theta)]=0$ より $\E[U]\approx\theta V$．
よって $Z=U/\sqrt V$ の平均は $\theta\sqrt V\approx\theta\sqrt{Dp(1-p)}$，分散は近似的に1．
\end{proof}
いずれも $\HR$ が1に近いことと正規近似に基づく\textbf{近似}である．
\cref{thm:12-core}（両側）で
\[
|\log\HR|\sqrt{Dp(1-p)}=z_{\alpha/2}+z_\beta
\]
とおけば
\begin{equation}
D=\frac{(z_{\alpha/2}+z_\beta)^2}{p(1-p)(\log\HR)^2},\qquad p=\tfrac12\ \text{なら}\ D=\frac{4(z_{\alpha/2}+z_\beta)^2}{(\log\HR)^2}.\label{eq:12-schoenfeld}
\end{equation}
1:1 割付なら総イベント数は $D=2r$ で\cref{eq:12-exp}と一致する．

\paragraph{イベント数から症例数へ}
必要症例数は $N=D/\Prob(\text{イベント})$ で，イベント確率は登録期間・追跡期間・ハザードから求める．
\begin{supbox}[補足：一様登録の下でのイベント確率]
登録期間 $a$ に一様に登録し，最終登録後さらに $f$ だけ追跡する．ハザード $\lambda$ の指数分布なら，1人がイベントを観察される確率は
$\Prob=1-\dfrac{e^{-\lambda f}-e^{-\lambda(a+f)}}{\lambda a}$（$S(t)$ を追跡期間 $[f,a+f]$ で平均して1から引く）．
\end{supbox}
生存時間試験で「症例数を増やす」だけでなく「追跡を延ばす」ことで検出力が上がるのは，精度を決めるのがイベント数だからである．

\section{その他の考慮}\label{sec:12-other}
\begin{itemize}
\item \textbf{脱落}：脱落率 $w$ を見込むなら $n/(1-w)$．
\item \textbf{不均等割付}：$k:1$ 割付では，総数は均等割付の $(1+k)^2/(4k)$ 倍必要（\cref{eq:12-schoenfeld}の $1/\{p(1-p)\}$ で $p=k/(1+k)$ とおくと $(1+k)^2/k$，$p=1/2$ の4との比）．
\item \textbf{多重性}：複数の主要比較があれば $\alpha$ を分割して $z_{\alpha/(2m)}$ を用いる（Bonferroni，\cref{ch:14}）．
\item \textbf{中間解析}：群逐次デザインでは最大症例数がわずかに増える（\cref{ch:14}）．
\item \textbf{同等性・非劣性}：対立仮説の側が変わるので式が異なる（\cref{ch:13}）．
\end{itemize}

\section{仮定}\label{sec:12-assump}
\begin{itemize}
\item 検定統計量の正規近似が成り立つ程度の標本サイズ．
\item 分散・対照群のイベント率などの見積もりが正しいこと（外れると検出力が大きく変わる）．
\item 生存時間：比例ハザード，$\HR$ が1に近いこと（Schoenfeld の式），登録・追跡・脱落の見込み．
\end{itemize}

\section{結果の解釈}\label{sec:12-interp}
\begin{itemize}
\item 「$n=122$ で検出力90\%」は，「真の差が $\Delta$ なら90\%の確率で有意になる」の意味．真の差が小さければ検出力は下がる．
\item 試験後に観測された効果量で「事後の検出力」を計算しても，P 値と1対1に対応するだけで新しい情報は与えない．非有意の解釈は信頼区間で行う．
\end{itemize}

\section{手法選択}\label{sec:12-choice}
主解析で用いる検定と同じ統計量・同じ尺度で症例数を計算する．
平均の比較なら\cref{eq:12-means}，比率なら\cref{eq:12-prop}，生存時間ならイベント数（\cref{eq:12-exp}または\cref{eq:12-schoenfeld}）から症例数へ，
順位和検定なら $\Prob(X>Y)$ を用いた\cref{eq:03-power}．
同じ仮説でも，統計量の尺度（例えば $\hat\lambda$ そのものか $\log\hat\lambda$ か）や情報量を評価する点（$\theta_0$ か $\theta_1$ か）が違えば正規近似の精度が違い，得られる症例数も変わる．
したがって「どの検定を用いるか」を先に決め，その検定に合わせた式を使う．

\section{よくある誤り}\label{sec:12-err}
\begin{warnbox}
\begin{itemize}
\item 両側検定なのに $z_\alpha$（片側）を使う，またはその逆．$z_\beta$ と $z_{\beta/2}$ を混同する．
\item 平均の差の式で係数2（2群分の分散）を落とす．結果を「総数」と「各群」で取り違える．
\item 生存時間で症例数とイベント数を混同する．\cref{eq:12-exp}は各群，\cref{eq:12-schoenfeld}は総数である．
\item 端数を切り捨てる（必ず切り上げる）．
\end{itemize}
\end{warnbox}

\section{数理統計との接続}\label{sec:12-link}
\begin{linkbox}
\begin{itemize}
\item 検出力関数（『現代数理統計学の基礎』定義7.13）と Neyman--Pearson の補題（定理7.16）．同書の $\beta(\theta)$ は検出力である．
\item \kakomonSM{2024}{1} では片側 $z$ 検定の検出力 $1-\Phi(z_\alpha-\beta_1/\SE)$ を2つの推定量で比べ，分散の小さい（有効な）推定量ほど検出力が高いことが問われた．
      症例数設計は「分散を小さくする」ことと「$n$ を増やす」ことが同じ効果をもつことの定量化である．
\item Schoenfeld の式の導出は，スコア検定の統計量 $U/\sqrt I$ と Wald 検定の統計量 $\hat\beta\sqrt I$ が $\hat\beta\approx U/I$ により漸近的に同等であることの応用である．
\end{itemize}
\end{linkbox}

\section{例題}\label{sec:12-ex}
\begin{reidai}[平均の差・比率の差]\label{reidai:12-1}
両側 $\alpha=0.05$（$z_{0.025}=1.960$）とする．
\begin{enumerate}[label=(\arabic*)]
\item 主要評価項目の SD を12，臨床的に意味のある差を5とし，検出力90\%（$z_{0.10}=1.282$）で各群の症例数を求めよ．
\item 対照群の有効率0.45，試験群0.60を想定し，検出力80\%（$z_{0.20}=0.842$）で\cref{eq:12-prop}により各群の症例数を求めよ（$\sqrt{0.49875}=0.7062$，$\sqrt{0.4875}=0.6982$）．
\item 各群50人，効果量 $\Delta/\sigma=0.5$ の平均の比較の検出力を求めよ（$\Phi(0.54)=0.705$）．
\end{enumerate}
\end{reidai}

\begin{reidai}[イベント数]\label{reidai:12-3}
HR $=0.7$ を両側 $\alpha=0.05$，検出力90\%で検出したい（1:1割付，$\log0.7=-0.3567$，$z_{0.025}=1.960$，$z_{0.10}=1.282$）．
\begin{enumerate}[label=(\arabic*)]
\item ログランク検定を用いるとして，必要な総イベント数を求めよ．
\item 追跡期間中にイベントを起こす確率が両群平均で0.6と見込まれるとき，必要な総症例数を求めよ．
\item 脱落を10\%見込むとどうなるか．
\end{enumerate}
\end{reidai}

\section{解答}\label{sec:12-sol}
\begin{kaitou}{reidai:12-1}
(1) \cref{eq:12-means}より $n=2\times144\times(1.960+1.282)^2/25=288\times10.51/25=121.1$，各群122人．\\
(2) $\bar\pi=0.525$．$1.960\times0.7062=1.384$，$0.842\times0.6982=0.588$，和 $1.972$，2乗 $3.889$．$n=3.889/0.15^2=3.889/0.0225=172.8$ → 各群173人．\\
(3) $\mu=(\Delta/\sigma)\sqrt{n/2}=0.5\times\sqrt{25}=2.5$，検出力 $\Phi(2.5-1.96)=\Phi(0.54)=0.705$（反対側の棄却確率は無視）．
